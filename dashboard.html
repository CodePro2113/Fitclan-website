Attack enemy territories to expand your clan's empire

One-time setup required

Run this SQL in your Supabase project's SQL Editor to enable territory tracking:

create table territory_control (
  territory_id text primary key,
  clan_id uuid,
  clan_name text,
  captured_by uuid references auth.users(id),
  captured_at timestamptz default now()
);
alter table territory_control enable row level security;
create policy "read" on territory_control for select using (true);
create policy "insert" on territory_control for insert with check (auth.uid() = captured_by);
-- Only the current capturer can overwrite a territory (prevents rival clans
-- stealing each other's territory via direct API call without going through
-- the attack flow in the app).
create policy "update" on territory_control for update using (true) with check (auth.uid() = captured_by);
